# Pip / uv alternative to environment.yml (for a plain venv instead of conda).
#
#   macOS / Linux:
#     uv venv si_env --python 3.12        # (or: python3.12 -m venv si_env)
#     source si_env/bin/activate
#     uv pip install -r requirements.txt   # (or: pip install -r requirements.txt)
#
#   Windows (Command Prompt):
#     uv venv si_env --python 3.12
#     si_env\Scripts\activate
#     uv pip install -r requirements.txt
#
# Use Python 3.12 — it has pre-built wheels for every dependency below on both
# macOS and Windows (Python 3.13 lacks some, e.g. hdbscan, on Windows).

jupyterlab
ipywidgets
ipympl
hdbscan
pyqtgraph
PySide6<6.8
ephyviewer
spikeinterface-gui
