# Pip / uv alternative to environment.yml (for a plain venv instead of conda).
#
#   macOS / Linux:
#     uv venv si_env --python 3.12        # (or: python3.12 -m venv si_env)
#     source si_env/bin/activate
#     uv pip install -r requirements.txt   # (or: pip install -r requirements.txt)
#
#   Windows (Command Prompt):
#     uv venv si_env --python 3.12
#     si_env\Scripts\activate
#     uv pip install -r requirements.txt
#
# Use Python 3.12 — it has pre-built wheels for every dependency below on both
# macOS and Windows, so pip never has to compile from source (which on Windows
# would need the MSVC C++ Build Tools). 3.13 has broader wheel coverage than it
# used to, but some deps can still lag, so 3.12 is the tested choice.
spikeinterface[full,widgets]
rich                # structured/coloured terminal output in scripts/run_sorting.py
prompt_toolkit      # arrow-key menu navigation in SpikeInterface_Menu.py (cross-platform)
jupyterlab
ipywidgets
ipympl
hdbscan
pyqtgraph
PySide6<6.8
ephyviewer
spikeinterface-gui
plotly>=5.20,<6
